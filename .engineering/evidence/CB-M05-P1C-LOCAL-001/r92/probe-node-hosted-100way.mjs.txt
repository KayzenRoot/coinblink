// THROWAWAY diagnostic probe (R92). Not part of the PR; deleted before commit.
import { mkdtempSync, readFileSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
import { convertV4MiniflareOptions, Miniflare, NoOpLog } from 'miniflare';
import { splitLocalSchemaStatements } from './test/modules/CB-M05/p1c-local-preview.mjs';
import { createLocalCmsHandler, issueBootstrapTicket, webauthnChallengeFromHex, CMS_TEST_ORIGIN } from './src/modules/CB-M05/p1c-local/local-cms.mjs';
import { makeSyntheticAuthenticator, makeRegistrationResponse } from './src/modules/CB-M05/p1b-local/webauthn-fixtures.mjs';
import { randomBytes, randomUUID } from 'node:crypto';

const RAW = Symbol('raw');
const N = Number(process.argv[2] ?? 100);
const TRIALS = Number(process.argv[3] ?? 3);
const schema = readFileSync('test/modules/CB-M05/p1c-fixtures/schema.sql', 'utf8');

function instrument(db, log) {
  const wrapStmt = (stmt, sql) => new Proxy(stmt, {
    get(target, prop) {
      if (prop === RAW) return target;
      if (prop === 'bind') return (...args) => wrapStmt(target.bind(...args), sql);
      if (['first', 'run', 'all', 'raw'].includes(prop)) {
        return async (...args) => {
          const t0 = performance.now();
          try { return await target[prop](...args); } catch (error) {
            log.push({ stage: sql.trim().split(/\s+/).slice(0, 4).join(' '), method: prop, ms: Math.round(performance.now() - t0),
              name: error?.name, msg: String(error?.message).slice(0, 120), cause: error?.cause?.code ?? String(error?.cause?.message ?? '').slice(0, 80) });
            throw error;
          }
        };
      }
      const v = target[prop];
      return typeof v === 'function' ? v.bind(target) : v;
    },
  });
  return new Proxy(db, {
    get(target, prop) {
      if (prop === 'prepare') return sql => wrapStmt(target.prepare(sql), sql);
      if (prop === 'batch') return async stmts => {
        const t0 = performance.now();
        try { return await target.batch(stmts.map(s => s[RAW] ?? s)); } catch (error) {
          if (log.length === 0) console.log('FIRST-BATCH-ERROR', JSON.stringify({ name: error?.name, msg: error?.message, cause: error?.cause && { code: error.cause.code, errno: error.cause.errno, syscall: error.cause.syscall, address: error.cause.address, port: error.cause.port, message: error.cause.message } }));
          log.push({ stage: 'BATCH', method: 'batch', ms: Math.round(performance.now() - t0), name: error?.name,
            msg: String(error?.message).slice(0, 120), cause: error?.cause?.code ?? String(error?.cause?.message ?? '').slice(0, 80) });
          throw error;
        }
      };
      const v = target[prop];
      return typeof v === 'function' ? v.bind(target) : v;
    },
  });
}

async function trial(t) {
  const dir = mkdtempSync(path.join(tmpdir(), 'r92-probe-'));
  const mf = new Miniflare(convertV4MiniflareOptions({
    name: 'r92-probe-' + randomUUID(),
    script: 'export default { async fetch() { return new Response("LOCAL ONLY") } }',
    modules: true, compatibilityDate: '2026-10-13',
    d1Databases: { CMS_D1_LOCAL: '00000000-0000-4000-8000-000000000086' },
    resourcePersistencePath: dir, log: new NoOpLog(),
  }));
  try {
    const raw = await mf.getD1Database('CMS_D1_LOCAL');
    for (const s of splitLocalSchemaStatements(schema)) await raw.prepare(s).run();
    const log = [];
    const db = instrument(raw, log);
    const secret = randomBytes(32).toString('hex');
    const env = { CB_M05_LOCAL_TEST_ONLY: 'true', CMS_D1_LOCAL: db, CMS_TEST_HMAC_KEY: secret };
    const handle = createLocalCmsHandler({ now: () => Date.now() });
    const req = (route, body) => new Request(CMS_TEST_ORIGIN + route, {
      method: 'POST', headers: { origin: CMS_TEST_ORIGIN, 'sec-fetch-site': 'same-origin', 'content-type': 'application/json' },
      body: JSON.stringify(body),
    });
    const set = [];
    for (let i = 0; i < N; i++) {
      const ticket = await issueBootstrapTicket(raw, secret, Math.floor(Date.now() / 1000));
      const authenticator = makeSyntheticAuthenticator('p-' + t + '-' + i);
      const r = await handle(req('/local/bootstrap/challenge', { ticket }), env);
      const { challenge } = await r.json();
      set.push({ ticket, authenticator, challenge });
    }
    const t0 = performance.now();
    const results = await Promise.all(set.map(async c => {
      const s = performance.now();
      const res = await handle(req('/local/bootstrap/claim', {
        ticket: c.ticket, challenge: c.challenge,
        response: makeRegistrationResponse(c.authenticator, { challenge: webauthnChallengeFromHex(c.challenge) }),
      }), env);
      return { status: res.status, ms: Math.round(performance.now() - s) };
    }));
    const wall = Math.round(performance.now() - t0);
    const hist = {};
    for (const r of results) hist[r.status] = (hist[r.status] ?? 0) + 1;
    const owners = (await raw.prepare('SELECT COUNT(*) AS n FROM cms_owner').first()).n;
    const creds = (await raw.prepare('SELECT COUNT(*) AS n FROM cms_credentials').first()).n;
    const cons = (await raw.prepare('SELECT COUNT(*) AS n FROM cms_bootstrap_tickets WHERE consumed_at IS NOT NULL').first()).n;
    const errs = {};
    for (const e of log) { const k = JSON.stringify([e.stage, e.method, e.name, e.msg, e.cause]); errs[k] = (errs[k] ?? 0) + 1; }
    console.log(`trial ${t}: N=${N} wall=${wall}ms hist=${JSON.stringify(hist)} owners=${owners} creds=${creds} consumedTickets=${cons}`);
    for (const [k, v] of Object.entries(errs)) console.log(`   x${v} ${k}`);
    const maxMs = Math.max(...results.map(r => r.ms));
    console.log(`   per-request max=${maxMs}ms`);
  } finally {
    await mf.dispose();
    rmSync(dir, { recursive: true, force: true });
  }
}

for (let t = 1; t <= TRIALS; t++) await trial(t);
