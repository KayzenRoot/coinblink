// THROWAWAY feasibility probe (R92): handler hosted INSIDE workerd, claims dispatched concurrently.
import { mkdtempSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
import { build } from 'esbuild';
import { convertV4MiniflareOptions, Miniflare, NoOpLog } from 'miniflare';
import { randomBytes, randomUUID } from 'node:crypto';
import { readFileSync } from 'node:fs';
import { splitLocalSchemaStatements } from './test/modules/CB-M05/p1c-local-preview.mjs';
import { issueBootstrapTicket, webauthnChallengeFromHex, CMS_TEST_ORIGIN } from './src/modules/CB-M05/p1c-local/local-cms.mjs';
import { makeSyntheticAuthenticator, makeRegistrationResponse } from './src/modules/CB-M05/p1b-local/webauthn-fixtures.mjs';

const N = Number(process.argv[2] ?? 100);
const TRIALS = Number(process.argv[3] ?? 2);
const schema = readFileSync('test/modules/CB-M05/p1c-fixtures/schema.sql', 'utf8');

const bundle = await build({
  stdin: {
    contents: "import { createLocalCmsHandler } from './src/modules/CB-M05/p1c-local/local-cms.mjs';\nconst handler = createLocalCmsHandler();\nexport default { fetch(request, env) { return handler(request, env); } };",
    resolveDir: process.cwd(), loader: 'js', sourcefile: 'entry.mjs',
  },
  bundle: true, format: 'esm', platform: 'neutral', write: false, target: 'es2022',
  conditions: ['workerd', 'worker', 'browser'], mainFields: ['module', 'main'],
});
const script = bundle.outputFiles[0].text;

async function trial(t) {
  const dir = mkdtempSync(path.join(tmpdir(), 'r92-probe3-'));
  const secret = randomBytes(32).toString('hex');
  const mf = new Miniflare(convertV4MiniflareOptions({
    name: 'r92-probe3-' + randomUUID(),
    script, modules: true, compatibilityDate: '2026-10-13', compatibilityFlags: ['nodejs_compat'],
    d1Databases: { CMS_D1_LOCAL: '00000000-0000-4000-8000-000000000086' },
    bindings: { CB_M05_LOCAL_TEST_ONLY: 'true', CMS_TEST_HMAC_KEY: secret },
    resourcePersistencePath: dir, log: new NoOpLog(),
  }));
  try {
    const raw = await mf.getD1Database('CMS_D1_LOCAL');
    for (const s of splitLocalSchemaStatements(schema)) await raw.prepare(s).run();
    const req = (route, body) => [CMS_TEST_ORIGIN + route, {
      method: 'POST', headers: { origin: CMS_TEST_ORIGIN, 'sec-fetch-site': 'same-origin', 'content-type': 'application/json' },
      body: JSON.stringify(body),
    }];
    const set = [];
    for (let i = 0; i < N; i++) {
      const ticket = await issueBootstrapTicket(raw, secret, Math.floor(Date.now() / 1000));
      const authenticator = makeSyntheticAuthenticator('w-' + t + '-' + i);
      const res = await mf.dispatchFetch(...req('/local/bootstrap/challenge', { ticket }));
      const { challenge } = await res.json();
      set.push({ ticket, authenticator, challenge });
    }
    const t0 = performance.now();
    const results = await Promise.all(set.map(async c => {
      try {
        const res = await mf.dispatchFetch(...req('/local/bootstrap/claim', {
          ticket: c.ticket, challenge: c.challenge,
          response: makeRegistrationResponse(c.authenticator, { challenge: webauthnChallengeFromHex(c.challenge) }),
        }));
        return { status: res.status };
      } catch (error) {
        return { status: 'TRANSPORT:' + (error?.cause?.code ?? error?.message) };
      }
    }));
    const hist = {};
    for (const r of results) hist[r.status] = (hist[r.status] ?? 0) + 1;
    const owners = (await raw.prepare('SELECT COUNT(*) AS n FROM cms_owner').first()).n;
    const creds = (await raw.prepare('SELECT COUNT(*) AS n FROM cms_credentials').first()).n;
    console.log(`workerd trial ${t}: N=${N} wall=${Math.round(performance.now() - t0)}ms hist=${JSON.stringify(hist)} owners=${owners} creds=${creds}`);
  } finally {
    await mf.dispose();
    rmSync(dir, { recursive: true, force: true });
  }
}
for (let t = 1; t <= TRIALS; t++) await trial(t);
