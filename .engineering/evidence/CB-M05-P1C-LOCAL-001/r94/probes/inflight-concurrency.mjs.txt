// THROWAWAY CP1 diagnostic (R94): does dispatchFetch deliver concurrent in-flight requests to workerd?
import { convertV4MiniflareOptions, Miniflare, NoOpLog } from 'miniflare';
import { randomUUID } from 'node:crypto';

const N = Number(process.argv[2] ?? 50);
const mf = new Miniflare(convertV4MiniflareOptions({
  name: 'r94-inflight-' + randomUUID(),
  modules: true, compatibilityDate: '2026-10-13',
  script: `
    let inFlight = 0, maxInFlight = 0, total = 0;
    export default {
      async fetch(request) {
        if (new URL(request.url).pathname === '/report') {
          return Response.json({ maxInFlight, total });
        }
        inFlight++; total++;
        maxInFlight = Math.max(maxInFlight, inFlight);
        await new Promise(resolve => setTimeout(resolve, 40));
        inFlight--;
        return Response.json({ inFlight });
      },
    };
  `,
  log: new NoOpLog(),
}));
try {
  const results = await Promise.all(Array.from({ length: N }, () =>
    mf.dispatchFetch('https://owner.example.test/slow').then(r => r.json())));
  const observed = Math.max(...results.map(r => r.inFlight));
  const { maxInFlight, total } = await (await mf.dispatchFetch('https://owner.example.test/report')).json();
  console.log(`N=${N} total=${total} maxInFlightInWorker=${maxInFlight} maxObservedByCallers=${observed}`);
} finally {
  await mf.dispose();
}
