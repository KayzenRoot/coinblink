// THROWAWAY CP1 differential (R94): challenge quota, Node-hosted vs Worker-hosted.
// Redacted: statuses, counts and error classes only. No challenges, cookies or keys are logged.
import { mkdirSync, mkdtempSync, readFileSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
import { convertV4MiniflareOptions, Miniflare, NoOpLog } from 'miniflare';
import { randomUUID, randomBytes } from 'node:crypto';
import { build } from 'esbuild';
import { fileURLToPath } from 'node:url';
import { splitLocalSchemaStatements } from './test/modules/CB-M05/p1c-local-preview.mjs';
import { createLocalCmsHandler, CMS_TEST_ORIGIN } from './src/modules/CB-M05/p1c-local/local-cms.mjs';

const TRIALS = Number(process.argv[2] ?? 20);
const CONCURRENCY = Number(process.argv[3] ?? 10);
const schema = readFileSync('test/modules/CB-M05/p1c-fixtures/schema.sql', 'utf8');
const stateRoot = path.resolve('.wrangler/r94-probe-state');

async function seed(db) {
  const now = Math.floor(Date.now() / 1000);
  await db.prepare('INSERT INTO cms_owner(owner_id,created_at) VALUES (1,?)').bind(now).run();
  const credentialId = randomBytes(32).toString('base64url');
  await db.prepare('INSERT INTO cms_credentials(credential_id,owner_id,public_key,sign_count,device_kind) VALUES (?,1,?,0,?)')
    .bind(credentialId, Buffer.alloc(65, 4), 'primary').run();
  return credentialId;
}

function histogram(statuses) {
  return statuses.reduce((counts, status) => ({ ...counts, [status]: (counts[status] ?? 0) + 1 }), {});
}

async function nodeHostedTrials(trials) {
  const dir = mkdtempSync(path.join(tmpdir(), 'r94-quota-node-'));
  const mf = new Miniflare(convertV4MiniflareOptions({
    name: 'r94-quota-node-' + randomUUID(),
    script: 'export default { async fetch() { return new Response("LOCAL ONLY") } }',
    modules: true, compatibilityDate: '2026-10-13',
    d1Databases: { CMS_D1_LOCAL: '00000000-0000-4000-8000-000000000086' },
    resourcePersistencePath: dir, log: new NoOpLog(),
  }));
  const summary = { match: 0, mismatch: 0, ackGT: 0, ackLT: 0 };
  try {
    const db = await mf.getD1Database('CMS_D1_LOCAL');
    for (const s of splitLocalSchemaStatements(schema)) await db.prepare(s).run();
    const credentialId = await seed(db);
    const env = { CB_M05_LOCAL_TEST_ONLY: 'true', CMS_D1_LOCAL: db, CMS_TEST_HMAC_KEY: randomBytes(32).toString('hex') };
    const handle = createLocalCmsHandler({ now: () => Date.now() });
    const ask = () => handle(new Request(CMS_TEST_ORIGIN + '/local/challenge', {
      method: 'POST', headers: { origin: CMS_TEST_ORIGIN, 'sec-fetch-site': 'same-origin', 'content-type': 'application/json' },
      body: JSON.stringify({ credentialId }),
    }), env);
    for (let t = 1; t <= trials; t++) {
      const now = Math.floor(Date.now() / 1000);
      await db.prepare('DELETE FROM cms_challenges').run();
      const results = await Promise.all(Array.from({ length: CONCURRENCY }, ask));
      const statuses = results.map(r => r.status);
      const ack = statuses.filter(s => s === 200).length;
      const rows = (await db.prepare(
        'SELECT COUNT(*) AS n FROM cms_challenges WHERE credential_id=? AND consumed_at IS NULL AND expires_at>?'
      ).bind(credentialId, now).first()).n;
      const verdict = ack === rows ? 'MATCH' : 'MISMATCH';
      if (verdict === 'MATCH') summary.match++; else summary.mismatch++;
      if (ack > rows) summary.ackGT++;
      if (ack < rows) summary.ackLT++;
      const consumed = (await db.prepare('SELECT COUNT(*) AS n FROM cms_challenges WHERE consumed_at IS NOT NULL').first()).n;
      const expired = (await db.prepare('SELECT COUNT(*) AS n FROM cms_challenges WHERE consumed_at IS NULL AND expires_at<=?').bind(now).first()).n;
      console.log(`node trial ${t}: statuses=[${statuses.join(',')}] ack=${ack} active=${rows} consumed=${consumed} expired=${expired} ${verdict}`);
    }
  } finally {
    await mf.dispose();
    rmSync(dir, { recursive: true, force: true });
  }
  console.log('node SUMMARY ' + JSON.stringify(summary));
  return summary;
}

async function workerHostedTrials(trials) {
  mkdirSync(stateRoot, { recursive: true });
  const script = (await build({
    entryPoints: [fileURLToPath(new URL('./test/modules/CB-M05/p1c-fixtures/bootstrap-worker-entry.mjs', import.meta.url))],
    bundle: true, format: 'esm', platform: 'neutral', target: 'es2022', write: false, logLevel: 'silent',
    conditions: ['workerd', 'worker', 'browser'], mainFields: ['module', 'main'],
  })).outputFiles[0].text;
  const secret = randomBytes(32).toString('hex');
  const mf = new Miniflare(convertV4MiniflareOptions({
    name: 'r94-quota-worker-' + randomUUID(),
    script, modules: true, compatibilityDate: '2026-10-13', compatibilityFlags: ['nodejs_compat'],
    d1Databases: { CMS_D1_LOCAL: '00000000-0000-4000-8000-000000000086' },
    bindings: { CB_M05_LOCAL_TEST_ONLY: 'true', CMS_TEST_HMAC_KEY: secret },
    resourcePersistencePath: path.join(stateRoot, randomUUID()), log: new NoOpLog(),
  }));
  const summary = { match: 0, mismatch: 0, ackGT: 0, ackLT: 0, workerD1Failures: 0 };
  try {
    const db = await mf.getD1Database('CMS_D1_LOCAL');
    for (const s of splitLocalSchemaStatements(schema)) await db.prepare(s).run();
    const credentialId = await seed(db);
    const dispatch = () => mf.dispatchFetch(CMS_TEST_ORIGIN + '/local/challenge', {
      method: 'POST', headers: { origin: CMS_TEST_ORIGIN, 'sec-fetch-site': 'same-origin', 'content-type': 'application/json' },
      body: JSON.stringify({ credentialId }),
    });
    for (let t = 1; t <= trials; t++) {
      const now = Math.floor(Date.now() / 1000);
      await db.prepare('DELETE FROM cms_challenges').run();
      const results = await Promise.all(Array.from({ length: CONCURRENCY }, dispatch));
      const statuses = results.map(r => r.status);
      const ack = statuses.filter(s => s === 200).length;
      const rows = (await db.prepare(
        'SELECT COUNT(*) AS n FROM cms_challenges WHERE credential_id=? AND consumed_at IS NULL AND expires_at>?'
      ).bind(credentialId, now).first()).n;
      const verdict = ack === rows ? 'MATCH' : 'MISMATCH';
      if (verdict === 'MATCH') summary.match++; else summary.mismatch++;
      if (ack > rows) summary.ackGT++;
      if (ack < rows) summary.ackLT++;
      if (verdict === 'MISMATCH') console.log(`worker trial ${t}: hist=${JSON.stringify(histogram(statuses))} ack=${ack} rows=${rows} ${verdict}`);
    }
    const { entries } = await (await mf.dispatchFetch(CMS_TEST_ORIGIN + '/__test/diagnostics')).json();
    summary.workerD1Failures = entries.length;
  } finally {
    await mf.dispose();
    rmSync(path.join(stateRoot), { recursive: true, force: true });
  }
  console.log('worker SUMMARY ' + JSON.stringify(summary));
  return summary;
}

await nodeHostedTrials(TRIALS);
await workerHostedTrials(TRIALS);
