// THROWAWAY CP1 diagnostic (R94): same-ticket batch contention inside workerd.
// Measures whether the batch-level (409) conflict path is exercised at higher contender counts.
import { mkdirSync, readFileSync, rmSync } from 'node:fs';
import path from 'node:path';
import { convertV4MiniflareOptions, Miniflare, NoOpLog } from 'miniflare';
import { randomUUID, randomBytes } from 'node:crypto';
import { build } from 'esbuild';
import { fileURLToPath } from 'node:url';
import { splitLocalSchemaStatements } from './test/modules/CB-M05/p1c-local-preview.mjs';
import { issueBootstrapTicket, webauthnChallengeFromHex, CMS_TEST_ORIGIN } from './src/modules/CB-M05/p1c-local/local-cms.mjs';
import { makeSyntheticAuthenticator, makeRegistrationResponse } from './src/modules/CB-M05/p1b-local/webauthn-fixtures.mjs';

const CONTENDERS = Number(process.argv[2] ?? 50);
const TRIALS = Number(process.argv[3] ?? 3);
const schema = readFileSync('test/modules/CB-M05/p1c-fixtures/schema.sql', 'utf8');
const stateRoot = path.resolve('.wrangler/r94-probe-state');

const script = (await build({
  entryPoints: [fileURLToPath(new URL('./test/modules/CB-M05/p1c-fixtures/bootstrap-worker-entry.mjs', import.meta.url))],
  bundle: true, format: 'esm', platform: 'neutral', target: 'es2022', write: false, logLevel: 'silent',
  conditions: ['workerd', 'worker', 'browser'], mainFields: ['module', 'main'],
})).outputFiles[0].text;

function histogram(statuses) {
  return statuses.reduce((counts, status) => ({ ...counts, [status]: (counts[status] ?? 0) + 1 }), {});
}

for (let trial = 1; trial <= TRIALS; trial++) {
  mkdirSync(stateRoot, { recursive: true });
  const secret = randomBytes(32).toString('hex');
  const mf = new Miniflare(convertV4MiniflareOptions({
    name: 'r94-same-' + randomUUID(),
    script, modules: true, compatibilityDate: '2026-10-13', compatibilityFlags: ['nodejs_compat'],
    d1Databases: { CMS_D1_LOCAL: '00000000-0000-4000-8000-000000000086' },
    bindings: { CB_M05_LOCAL_TEST_ONLY: 'true', CMS_TEST_HMAC_KEY: secret },
    resourcePersistencePath: path.join(stateRoot, randomUUID()), log: new NoOpLog(),
  }));
  try {
    const db = await mf.getD1Database('CMS_D1_LOCAL');
    for (const s of splitLocalSchemaStatements(schema)) await db.prepare(s).run();
    const ticket = await issueBootstrapTicket(db, secret, Math.floor(Date.now() / 1000));
    const post = (route, body) => mf.dispatchFetch(CMS_TEST_ORIGIN + route, {
      method: 'POST', headers: { origin: CMS_TEST_ORIGIN, 'sec-fetch-site': 'same-origin', 'content-type': 'application/json' },
      body: JSON.stringify(body),
    });
    const contenders = [];
    for (let i = 0; i < CONTENDERS; i++) {
      const authenticator = makeSyntheticAuthenticator('same-' + trial + '-' + i);
      const res = await post('/local/bootstrap/challenge', { ticket });
      const { challenge } = await res.json();
      contenders.push({ authenticator, challenge });
    }
    const outcomes = await Promise.all(contenders.map(c => post('/local/bootstrap/claim', {
      ticket, challenge: c.challenge,
      response: makeRegistrationResponse(c.authenticator, { challenge: webauthinHex(c.challenge) }),
    })));
    const statuses = outcomes.map(r => r.status);
    const owners = (await db.prepare('SELECT COUNT(*) AS n FROM cms_owner').first()).n;
    const creds = (await db.prepare('SELECT COUNT(*) AS n FROM cms_credentials').first()).n;
    console.log(`same trial ${trial}: K=${CONTENDERS} hist=${JSON.stringify(histogram(statuses))} owners=${owners} creds=${creds} ` +
      `conflicts409=${statuses.filter(s => s === 409).length}`);
  } finally {
    await mf.dispose();
    rmSync(path.join(stateRoot), { recursive: true, force: true });
  }
}

function webauthinHex(hex) { return webauthnChallengeFromHex(hex); }
